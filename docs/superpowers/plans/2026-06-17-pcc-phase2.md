// AstroLab — cone search Gaia DR3 via Vizier (asu-tsv) para a PCC. Online, sem auth.
// Devolve [] em qualquer falha; nunca lança. Ver design 2026-06-17 (PCC Fase 2).
using System.Globalization;

namespace AstroLab.Services;

public sealed record GaiaStar(double Ra, double Dec, double GMag, double BpRp);

public static class GaiaCatalog
{
    static readonly HttpClient Http = new() { Timeout = TimeSpan.FromSeconds(20) };

    /// <summary>Parseia a resposta TSV do Vizier. Linha de dados = 1º campo é double
    /// e há ≥4 campos com BP-RP parseável. Ignora meta (#), cabeçalho, unidades e separador.</summary>
    public static List<GaiaStar> ParseTsv(string tsv)
    {
        var stars = new List<GaiaStar>();
        foreach (var raw in tsv.Replace("\r", "").Split('\n'))
        {
            if (raw.Length == 0 || raw[0] == '#') continue;
            var f = raw.Split('\t');
            if (f.Length < 4) continue;
            if (!double.TryParse(f[0].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out var ra)) continue;
            if (!double.TryParse(f[1].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out var dec)) continue;
            if (!double.TryParse(f[2].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out var g)) continue;
            if (!double.TryParse(f[3].Trim(), NumberStyles.Float, CultureInfo.InvariantCulture, out var bpRp)) continue;
            stars.Add(new GaiaStar(ra, dec, g, bpRp));
        }
        return stars;
    }

    public static IReadOnlyList<GaiaStar> Query(double raDeg, double decDeg, double radiusDeg, out string status)
    {
        try
        {
            string c = Uri.EscapeDataString(
                $"{raDeg.ToString("0.######", CultureInfo.InvariantCulture)} {decDeg.ToString("0.######", CultureInfo.InvariantCulture)}");
            string url = "https://vizier.cds.unistra.fr/viz-bin/asu-tsv?-source=I/355/gaiadr3"
                       + $"&-c={c}&-c.rd={radiusDeg.ToString("0.####", CultureInfo.InvariantCulture)}"
                       + "&-out=RA_ICRS,DE_ICRS,Gmag,BP-RP&-out.max=5000&-sort=Gmag";
            string tsv = Http.GetStringAsync(url).GetAwaiter().GetResult();
            var stars = ParseTsv(tsv);
            status = stars.Count > 0 ? $"Gaia: {stars.Count} estrelas" : "Gaia: 0 estrelas";
            return stars;
        }
        catch (Exception ex) { status = $"Gaia indisponível: {ex.Message}"; return Array.Empty<GaiaStar>(); }
    }
}
```

// AstroLab — calibração fotométrica de cor (PCC). Regressão BP-RP sobre estrelas
// Gaia medidas na imagem linear → ganhos por canal. Ver design 2026-06-17 (Fase 2).
namespace AstroLab.Services;

public static class PhotometricCalibration
{
    public sealed record StarSample(double R, double G, double B, double BpRp);

    /// <summary>Ajusta log(R/G) e log(B/G) vs BP-RP e devolve os ganhos (gR,gB) que
    /// neutralizam uma estrela do ponto branco. gG=1 implícito.</summary>
    public static (double gR, double gB) FitGains(IReadOnlyList<StarSample> stars, double whiteBpRp)
    {
        var xs = new List<double>(stars.Count);
        var yR = new List<double>(stars.Count);
        var yB = new List<double>(stars.Count);
        foreach (var s in stars)
        {
            if (s.R <= 0 || s.G <= 0 || s.B <= 0) continue;
            xs.Add(s.BpRp);
            yR.Add(Math.Log(s.R / s.G));
            yB.Add(Math.Log(s.B / s.G));
        }
        var (aR, bR) = LinFit(xs, yR);
        var (aB, bB) = LinFit(xs, yB);
        double gR = Math.Exp(-(aR + bR * whiteBpRp));
        double gB = Math.Exp(-(aB + bB * whiteBpRp));
        return (gR, gB);
    }

    /// <summary>Mínimos quadrados y = a + b·x. Se variância de x ~0, b=0 (a=média de y).</summary>
    static (double a, double b) LinFit(List<double> x, List<double> y)
    {
        int n = x.Count;
        if (n == 0) return (0, 0);
        double sx = 0, sy = 0, sxx = 0, sxy = 0;
        for (int i = 0; i < n; i++) { sx += x[i]; sy += y[i]; sxx += x[i] * x[i]; sxy += x[i] * y[i]; }
        double denom = n * sxx - sx * sx;
        if (Math.Abs(denom) < 1e-12) return (sy / n, 0);
        double b = (n * sxy - sx * sy) / denom;
        double a = (sy - b * sx) / n;
        return (a, b);
    }
}
```

```csharp
    public sealed record PccSettings(int MinStars = 30, int ApertureRadiusPx = 6, double WhitePointBpRp = 0.82);

    /// <summary>Calibração fotométrica completa. Query Gaia → projeta → mede R/G/B →
    /// seleciona → FitGains → aplica. Aplica in-place; false+status em falha. Nunca lança.</summary>
    public static bool TryCalibrate(LinearImage img, WcsSolution wcs, PccSettings s, out string status)
    {
        try
        {
            int W = img.Width, H = img.Height;
            double radius = 0.5 * Math.Sqrt(
                Math.Pow(wcs.FovWidthDeg(W), 2) + Math.Pow(wcs.FovHeightDeg(H), 2));
            var cat = GaiaCatalog.Query(wcs.CenterRaDeg, wcs.CenterDecDeg, radius, out var gaiaStatus);
            if (cat.Count == 0) { status = gaiaStatus; return false; }

            // projetar para píxeis (0-based) e guardar candidatos dentro da imagem
            int ap = s.ApertureRadiusPx, margin = ap + 3;
            var px = new (double x, double y, double bpRp, double gmag)[cat.Count];
            int nIn = 0;
            foreach (var st in cat)
            {
                var (fx, fy) = wcs.WorldToPixel(st.Ra, st.Dec);
                double x = fx - 1, y = fy - 1;
                if (st.GMag < 7 || st.GMag > 15) continue;       // gama útil
                if (x < margin || y < margin || x >= W - margin || y >= H - margin) continue;
                px[nIn++] = (x, y, st.BpRp, st.GMag);
            }

            var samples = new List<StarSample>();
            for (int i = 0; i < nIn; i++)
            {
                var (x, y, bpRp, _) = px[i];
                // isolamento: sem outro candidato a < 3·ap
                bool isolated = true;
                for (int j = 0; j < nIn && isolated; j++)
                    if (j != i)
                    {
                        double dx = px[j].x - x, dy = px[j].y - y;
                        if (dx * dx + dy * dy < (3.0 * ap) * (3.0 * ap)) isolated = false;
                    }
                if (!isolated) continue;

                if (!Aperture(img, (int)Math.Round(x), (int)Math.Round(y), ap, out double r, out double g, out double b))
                    continue;                                    // saturada ou inválida
                samples.Add(new StarSample(r, g, b, bpRp));
            }

            if (samples.Count < s.MinStars) { status = $"PCC: poucas estrelas ({samples.Count})"; return false; }

            var (gR, gB) = FitGains(samples, s.WhitePointBpRp);
            ApplyGains(img, gR, gB);
            status = $"PCC ok — {samples.Count} estrelas, gR={gR:F2} gB={gB:F2}";
            return true;
        }
        catch (Exception ex) { status = $"PCC erro: {ex.Message}"; return false; }
    }

    /// <summary>Soma de abertura por canal menos fundo local (anel). false se saturada
    /// (pico ≥ 0.95 em qualquer canal) ou fluxo não positivo.</summary>
    static bool Aperture(LinearImage img, int cx, int cy, int ap, out double r, out double g, out double b)
    {
        r = g = b = 0;
        int W = img.Width; var d = img.Data;
        double[] sum = new double[3]; int nAp = 0;
        double[] bg = new double[3]; int nBg = 0;
        float peak = 0;
        int outer = ap + 4;
        for (int dy = -outer; dy <= outer; dy++)
            for (int dx = -outer; dx <= outer; dx++)
            {
                int x = cx + dx, y = cy + dy;
                int idx = (y * W + x) * 3;
                double rr = dx * dx + dy * dy;
                if (rr <= ap * ap)
                {
                    for (int c = 0; c < 3; c++) { sum[c] += d[idx + c]; if (d[idx + c] > peak) peak = d[idx + c]; }
                    nAp++;
                }
                else if (rr <= outer * outer)
                {
                    for (int c = 0; c < 3; c++) bg[c] += d[idx + c];
                    nBg++;
                }
            }
        if (peak >= 0.95f || nAp == 0 || nBg == 0) return false;
        r = sum[0] - bg[0] / nBg * nAp;
        g = sum[1] - bg[1] / nBg * nAp;
        b = sum[2] - bg[2] / nBg * nAp;
        return r > 0 && g > 0 && b > 0;
    }

    /// <summary>Aplica ganhos por canal (gG=1), estrutura do ColorCalibrate:
    /// subtrai mediana de fundo por canal, multiplica, repõe o fundo do G.</summary>
    static void ApplyGains(LinearImage img, double gR, double gB)
    {
        int N = img.Width * img.Height; var d = img.Data;
        double[] g = { gR, 1.0, gB };
        var bg = new float[3];
        for (int c = 0; c < 3; c++)
        {
            var ch = new float[N];
            for (int i = 0; i < N; i++) ch[i] = d[i * 3 + c];
            bg[c] = AstroPipeline.MedianOf(ch);
        }
        for (int c = 0; c < 3; c++)
            for (int i = 0; i < N; i++)
                d[i * 3 + c] = Math.Clamp((float)((d[i * 3 + c] - bg[c]) * g[c] + bg[1]), 0f, 1f);
    }
```

    {
        Console.WriteLine($"== PCC end-to-end (A=halos, B=PCC): {path} ==");
        var baseImg = TiffIO.LoadFloat(path);
        AstroPipeline.Normalize(baseImg);
        baseImg = AstroPipeline.Crop(baseImg, 0.012);

        var sw = Stopwatch.StartNew();
        bool solved = PlateSolve.TrySolve(baseImg, null, 247.0, 5.74, out var wcs, out var solveStatus);
        sw.Stop();
        Console.WriteLine($"  solve: {solveStatus} em {sw.ElapsedMilliseconds} ms");
        if (!solved || wcs is null) throw new Exception($"solve falhou: {solveStatus}");

        // A — método por halos
        var a = baseImg.Clone();
        AstroPipeline.ColorCalibrate(a);

        // B — PCC
        var b = baseImg.Clone();
        bool ok = PhotometricCalibration.TryCalibrate(b, wcs, new PhotometricCalibration.PccSettings(), out var pccStatus);
        Console.WriteLine($"  {pccStatus}");
        if (!ok) throw new Exception($"PCC falhou: {pccStatus}");

        // esticar ambos (defaults) e gravar para comparação
        var p = ToneParams.Defaults;
        foreach (var (img2, tag) in new[] { (a, "a_halos"), (b, "b_pcc") })
        {
            AstroPipeline.Stretch(img2, p);
            AstroPipeline.Scnr(img2, p.Scnr);
            AstroPipeline.SaturationAndCurve(img2, p.Saturation);
            File.WriteAllBytes($"testdata/pcc_{tag}.jpg", PreviewRenderer.EncodeJpeg(img2, 92));
        }
        Console.WriteLine("  -> testdata/pcc_a_halos.jpg / pcc_b_pcc.jpg");
    }
